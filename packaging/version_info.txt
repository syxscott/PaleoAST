# UTF-8
#
# Windows version resource for PaleoAST.
#
# Without this the exe reports no version at all: Windows shows 0.0.0.0 in
# the file properties, "Unknown" in the task manager, and the entry in
# Installed Apps has no version column -- which is the difference between a
# program and a piece of software.
#
# The numeric fields must be plain integers. PyInstaller parses them with
# int(), so a dotted "1.1.2" is invalid here even though that is the string
# used everywhere else; it lives in StringFileInfo instead. The patch field
# is the fourth number, so 1.1.2 is (1, 1, 2, 0).
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(1, 1, 2, 0),
    prodvers=(1, 1, 2, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0),
  ),
  kids=[
    StringFileInfo(
      [
        StringTable(
          "040904B0",
          [StringStruct("CompanyName", "PaleoAST Development Team"),
           StringStruct("FileDescription", "PaleoAST - Paleontological Advanced Statistical Toolkit"),
           StringStruct("FileVersion", "1.1.2"),
           StringStruct("InternalName", "PaleoAST"),
           StringStruct("LegalCopyright", "PaleoAST Development Team"),
           StringStruct("OriginalFilename", "PaleoAST.exe"),
           StringStruct("ProductName", "PaleoAST"),
           StringStruct("ProductVersion", "1.1.2")])
      ]),
    VarFileInfo([VarStruct("Translation", [1033, 1200])])
  ]
)
